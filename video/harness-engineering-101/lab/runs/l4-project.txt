# pager.py
"""Split a list of items into pages."""

def page_count(n_items, per_page):
    """How many pages are needed for n_items."""
    return n_items // per_page

def get_page(items, page, per_page):
    """Return the items on a 1-based page number."""
    start = (page - 1) * per_page
    return items[start:start + per_page]

# test_pager.py (테스트 셋)
from pager import get_page, page_count
class TestPager(unittest.TestCase):
    def test_exact_fit(self):
        self.assertEqual(page_count(20, 10), 2)
    def test_leftover_items_need_a_page(self):
        self.assertEqual(page_count(21, 10), 3)
    def test_last_page(self):
        self.assertEqual(get_page(list(range(21)), 3, 10), [20])
